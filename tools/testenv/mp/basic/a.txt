# Client A: does things; client B checks it sees them.
goto spawn
pickup off
wait 60
sync start
claimbrain                    # first of the team in: A is the brain
wait 60
expect kind brain
sync brains

# mining (server computes the pixel mask and broadcasts it)
hold CopperPickaxe                # (the server placed a dirt block at 2 -1)
aimtile 2 -1
use 2
wait 60
expect tile 2 -1 partial
sync mined

# a client calls down an empty ship, throws wood into it and sends it up: it reaches the shared stockpile
ship 5
wait 520
expect shipstate waiting
droptile Wood 60 5 -3
wait 90
expect shipcargo >= 60
shiplaunch
wait 450
expect stock Wood >= 60
sync launched

# a client-owned bomb
projectile Bomb -170 -16 0 0
wait 240
expect tile -10 1 open
sync bombed

# deploy a body
pickup on
give CloneBody 2
wait 10
pickup off
hold CloneBody
aim 90 -30
use 2
wait 60
expect units == 1
sync deployed
sync switched                 # B takes over that body
expect units == 1             # B's old body is now the idle one

# B's body is overkilled with a spare standing by (A is the brain: it can't transfer): everyone sees it burst
sync b-died
wait 20
expect deathfx gib            # broadcast: a control transfer isn't a vanilla death

# equipment is picked up deliberately (F): it disappears for everyone
spawnitem IronBroadsword 16 0
wait 60
sync sword-down
pickup
wait 30
expect holding IronBroadsword
sync sword-taken

# a ship ordered by A delivers wood next to A
ship 3 Wood 10
pickup on
wait 560
expect inv Wood >= 10
screenshot a-final
sync done

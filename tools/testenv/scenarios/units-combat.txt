# Idle bodies fight with the weapon they hold, and follow orders.
nospawn
goto spawn
flatten 60 14
wait 30
give FlintlockPistol
give MusketBall 100
hold FlintlockPistol
give CloneBody 2
hold CloneBody
aim 90 -30
use 2
wait 10
hold FlintlockPistol
aim 0 -400
switch                          # our pistol-holding body stays behind as an idle sentry
wait 10
expect units == 1
spawnnpc Zombie 200 -30
wait 600
expect npcs Zombie == 0         # the sentry shot it
screenshot sentry
order follow

# Idle bodies are the same CC actor: a long fall hurts them, a nearby blast knocks them flat.
order sentry
wait 30
expect unitlife >= 90
liftunit 20                     # loaded bodies can exceed the 4800 Ns gib limit at 25 tiles, depending on pose
wait 150
expect unitlife < 90
expect units == 1
heal
spawnnpc Zombie 0 0
clearnpcs
maxlife 400                     # our own body is near the blast too
wait 2
heal
placeunit 6
wait 10
blastunit -24                   # a blast 1.5 tiles to its left
wait 2
expect unitknocked yes
screenshot unit-knocked

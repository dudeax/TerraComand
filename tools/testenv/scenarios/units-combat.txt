# Idle bodies fight with the weapon they hold, and follow orders.
nospawn
goto spawn
flatten 60 14
wait 30
give FlintlockPistol
give MusketBall 100
hold FlintlockPistol
hold CloneBody
aim 90 -30
use 2
wait 10
hold FlintlockPistol
aim 0 -400
switch                          # our pistol-holding body stays behind as an idle sentry
wait 10
expect units == 1
spawnnpc Zombie 200 -30
wait 600
expect npcs Zombie == 0         # the sentry shot it
screenshot sentry
orders                          # follow

# Idle bodies follow the same physics: a long fall hurts them, a nearby blast knocks them flat.
orders                          # back to sentry
wait 30
expect unitlife >= 90
liftunit 40
wait 150
expect unitlife < 90
heal
spawnnpc Zombie 0 0
clearnpcs
maxlife 400                     # our own body is near the blast too
wait 2
heal
placeunit 6
wait 10
blastunit -24                   # a blast 1.5 tiles to its left
wait 2
expect unitknocked yes
screenshot unit-knocked

# Pickaxe power and hardness: copper can't carve Lihzahrd brick, but can carve dirt and stone.
# (Blocks sit on the ground: a CC digger digs the first solid thing along the line from its hands.)
nospawn
goto spawn
flatten 30 12
wait 30
settile 2 -1 LihzahrdBrick
settile -2 -1 Stone
hold CopperPickaxe
aimtile 2 -1
use 30
wait 5
expect tile 2 -1 solid
aimtile -2 -1
probe 0
use 2                          # one swing
wait 30
expect tile -2 -1 partial
use 60                         # vanilla: copper breaks stone in 3 hits
wait 5
expect tile -2 -1 empty
expect inv StoneBlock >= 1     # mining drops the block's item
settile -3 -1 Dirt              # inside the digger’s 56px reach from the physical hand
aimtile -3 -1
use 2
wait 30
expect tile -3 -1 partial       # dirt: 2 hits with copper
use 2
wait 30
expect tile -3 -1 empty

# Cortex Command style aiming and damage.
nospawn
clearnpcs                       # town NPCs shoot at test targets
goto spawn
flatten 60 24
wait 30
give FlintlockPistol
give MusketBall 999
give Minishark
hold FlintlockPistol
aim 300 -10

# CC firearms: spread per round shrinks with sharp aim (aiming far out, holding steady) and doubles unsupported (airborne)
aim 300 -10
wait 150                        # sharp aim builds up
expect sharpaim >= 0.9
mark
shoot 3 30
expect cone <= 0.016            # flintlock: 3 deg shake -> 1.65 sharp (half-range, radians)
aim 25 -2                       # aiming close in: no sharp aim
wait 60
expect sharpaim <= 0.05
mark
shoot 3 30
expect cone >= 0.022
expect cone <= 0.053             # physical stance can briefly lose support during a shot (CC x2 airborne spread)
goto rel 0 -20                  # falling: unsupported, x2 (arena is flattened 24 tiles high)
wait 12
mark
shoot 1 0
expect cone >= 0.04
wait 160
heal

# magazines: rounds count down; an empty gun reloads by itself; R reloads a part-empty one
aim 300 -10
reload
wait 80
expect magazine == 15
expect rounds == 15
wait 40
shoot 3 25
wait 30
expect rounds == 12
reload
wait 2
expect reloading yes
wait 80
expect reloading no
expect rounds == 15
hold Minishark
wait 5
expect magazine == 40
use 400                         # full auto: empties the magazine, then reloads by itself
wait 20
expect reloading yes
wait 120
expect rounds == 40
hold FlintlockPistol

# headshots on an upright target, body shots aren't
hold FlintlockPistol
spawnnpc Zombie 160 0
wait 60                         # let it land first
pin Zombie
hold down on
aimnpc Zombie body
shoot 1 30
wait 10
expect headshot no
expect wounds Zombie >= 1
screenshot wounds
aimnpc Zombie head
shoot 1 30
wait 10
expect headshot yes
hold down off

# no natural regen
hurt 40
wait 300
expect life <= 60
expect mywounds >= 1

# a bomb gibs a (pinned) zombie
mark
spawnnpc Zombie -120 0
wait 60
pin Zombie
projectile Bomb -120 -30 0 0 100
wait 200
expect gibs >= 1
